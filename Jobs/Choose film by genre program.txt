/* Temp file for API response */
filename omdbresp "/home/student/Data/omdb.json";

/* Call OMDb API */
proc http
  url="https://www.omdbapi.com/?apikey=&api_key&t=%sysfunc(urlencode(&title))"
  method="GET"
  out=omdbresp;
run;

/* Read JSON */
libname omdbjson json fileref=omdbresp;

/* Promote values to macro variables */
data job_data;
  set omdbjson.root;

    call symputx('mv_error', Error);
