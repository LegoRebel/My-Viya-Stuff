/* OMDb API key */
%let api_key = 61faee7;

/* Search text from Visual Analytics */
%put NOTE: Search term = &title;

/* Temp file for API response */
filename omdbresp "/home/student/Data/omdb_search.json";

/* Call OMDb SEARCH API (multi-row result) */
proc http
  url="https://www.omdbapi.com/?apikey=&api_key&s=%sysfunc(urlencode(&title))"
  method="GET"
  out=omdbresp;
run;

libname omdbjson json fileref=omdbresp;

ods _all_ close;


data sorted_film;
    set omdbjson.search;
    where upcase(type)="MOVIE";
run;

proc sort data =sorted_film nodupkey;
    by descending imdbid;
run;


data _null_;
  set sorted_film;
 
  link=cats("https://www.boxofficemojo.com//title/",imdbID,"/?ref_=bo_cso_ta");

  /* Container for one movie */
put '<div style="';

put 'background:white;';
put 'overflow:auto;';
put 'max-width:1000px;';
put '">';

/* Poster (clickable) */
if Poster ne "N/A" then do;
  put '<a href="' link +(-1) '" ';
  put 'target="_blank" rel="noopener noreferrer">';
  put '<img src="' Poster +(-1) '" ';
  put 'style="float:right;width:120px;margin-left:15px;cursor:pointer;">';
  put '</a>';

/* Title (clickable) */
put '<h3 style="margin-top:0;">';

put 'style="text-decoration:none;color:#0078D4;">';

/* Supporting information */
put '<p><strong>Type:</strong> ' Type '</p>';

/* User guidance text */
